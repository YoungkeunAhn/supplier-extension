// 테이블이 로드될 때까지 대기하는 함수
const waitForTable = (maxAttempts = 20, interval = 500) => {
  return new Promise((resolve, reject) => {
    let attempts = 0

    const checkTable = () => {
      attempts++
      console.log(`테이블 찾기 시도 ${attempts}/${maxAttempts}`)

      let table = document.querySelector('.react-bootstrap-table .table')

      if (!table) {
        table = document.querySelector('table.table')
      }

      if (!table) {
        table = document.querySelector('.table')
      }

      // 디버깅: 페이지의 모든 테이블 확인
      if (attempts === 1) {
        const allTables = document.querySelectorAll('table')
        console.log('페이지의 모든 테이블:', allTables)
        const reactBootstrap = document.querySelectorAll('.react-bootstrap-table')
        console.log('.react-bootstrap-table 요소:', reactBootstrap)
      }

      if (table) {
        console.log('테이블 발견!', table)
        resolve(table)
      } else if (attempts >= maxAttempts) {
        reject(new Error('테이블을 찾을 수 없습니다. 페이지를 확인해주세요.'))
      } else {
        setTimeout(checkTable, interval)
      }
    }

    checkTable()
  })
}

const getTotalPages = (items) => {
  const numberItems = Array.from(items)
    .map((el) => el.textContent.trim())
    .filter((txt) => /^\d+$/.test(txt)) // 정수 문자열만 남기기

  // 2️⃣ 마지막 숫자 li 추출
  const lastPageNumber = numberItems.length
    ? parseInt(numberItems[numberItems.length - 1], 10)
    : null

  return lastPageNumber
}

const sleep = (ms) => new Promise((res) => setTimeout(res, ms))
