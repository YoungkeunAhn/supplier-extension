document.addEventListener('DOMContentLoaded', () => {
  const getOrderBtn = document.getElementById('getOrderBtn')
  const getCenterBtn = document.getElementById('getCenterBtn')
  const q_date = document.getElementById('q_date')
  const q_date_change = document.getElementById('q_date_change')
  const q_stores = document.getElementById('q_stores')
  const fileInput = document.getElementById('fileInput')

  if (q_date) {
    q_date.value = dayjs().format('YYYY-MM-DD')
  }

  if (getOrderBtn) {
    getOrderBtn.addEventListener('click', async () => {

        alert('기존 입고예정일을 선택해주세요.')
        return
      }

      if (!auth) {
        alert('빅셀에 먼저 로그인해주세요.')
        return
      }

      const orders = await sendMsgOrders(auth, q_date.value, q_stores.value)
      const newOrders = transformOrders(orders)
      console.log('orders : ', newOrders)

      if (newOrders) {
        const res = await downloadOrdersExcel(newOrders)
        console.log('res : ', res)

        if (res.success && res.data) {
          downloadExcel(res)
        }
      }
    })

    if (getCenterBtn) {
      // await checkCoupangTab()
      getCenterBtn.addEventListener('click', async () => {
        if (!q_date_change.value) {
          alert('변경 입고예정일을 선택해주세요.')
          return
        }

        fileInput.click()
      })
    }

    if (fileInput) {
      fileInput.addEventListener('change', async (e) => {
        const file = e.target.files[0]

        if (!file) {
          alert('파일을 선택해주세요.')
          return
        }

        const reader = new FileReader()

        reader.onload = async (e) => {
          try {
            console.log('파일 읽기 시작')
            const arrayBuffer = e.target.result
            const workbook = new Excel.Workbook()
            await workbook.xlsx.load(arrayBuffer)

            const sheet = workbook.getWorksheet(1)
            const orders = []

            // 2행부터 데이터 읽기 (1행은 헤더)
            sheet.eachRow((row, rowNumber) => {
              if (rowNumber > 1) {
                // 헤더 제외
                const orderNo = row.getCell(1).value
                if (orderNo) {
                  orders.push(orderNo)
                }
              }
            })

            const uniqOrders = _.uniq(orders)

            console.log('uniqOrders : ', uniqOrders)

            await searchTableAndAddClick(uniqOrders)
          } catch (error) {
            console.error('파일 처리 오류:', error)
            alert('파일 처리 실패: ' + error.message)
          }
        }

        reader.onerror = (e) => {
          console.error('파일 읽기 오류:', e)
          alert('파일 읽기 실패')
        }

        reader.readAsArrayBuffer(file)
      })
    }
  }
})

const waitTabComplete = (id) =>
  new Promise((resolve) => {
    const listener = (tabId, info, tab) => {
      if (tabId === id && info.status === 'complete') {
        chrome.tabs.onUpdated.removeListener(listener)
        resolve(tab)
      }
    }
    chrome.tabs.onUpdated.addListener(listener)
  })

// 엑셀 다운로드
const downloadExcel = async (res) => {
  if (res?.success && res?.data) {
    // Base64를 Blob으로 변환
    const byteCharacters = atob(res.data)
    const byteNumbers = new Array(byteCharacters.length)
    for (let i = 0; i < byteCharacters.length; i++) {
      byteNumbers[i] = byteCharacters.charCodeAt(i)
    }
    const byteArray = new Uint8Array(byteNumbers)
    const blob = new Blob([byteArray], {
      type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    })

    // 다운로드 링크 생성 및 클릭
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = res.filename
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    URL.revokeObjectURL(url)

    // alert('엑셀 다운로드 완료!')
  } else {
    alert('엑셀 다운로드 실패: ' + (res.error || '알 수 없는 오류'))
  }
}

const toNum = (v) => Math.max(0, Number(v) || 0)
const transformOrders = (orders) => {
  const grouped = _.groupBy(orders, 'item_sku_id')
  const result = []

  _.forEach(grouped, (rows, sku) => {
    // 초기 본사/입고예정: 그룹 내 최댓값(스냅샷들이 섞여 있을 수 있으므로)
    const initStock = _.get(
      _.maxBy(rows, (r) => toNum(r.hq_stock_qty)),
      'hq_stock_qty',
      0
    )
    const initIncoming = _.get(
      _.maxBy(rows, (r) => toNum(r.hq_stock_pending_qty)),
      'hq_stock_pending_qty',
      0
    )

    let remainStock = toNum(initStock)
    let remainIncoming = toNum(initIncoming)

    // 처리 순서: 주문 일시가 있으면 그걸로, 없으면 입력 순서 유지
    const sorted = _.sortBy(rows, (r) => r.order_datetime ?? r.order_date ?? 0)

    sorted.forEach((row, idx) => {
      const need = toNum(row.order_qty)

      // 1) 본사 재고 우선 소진
      const useStock = Math.min(need, remainStock)
      remainStock -= useStock

      // 2) 입고예정으로 보충 소진
      const stillNeed = need - useStock
      const useIncoming = Math.min(stillNeed, remainIncoming)
      remainIncoming -= useIncoming

      const confirmed = useStock + useIncoming

      result.push({
        ...row,
        confirmed_qty: confirmed,

        allocated_from_stock: useStock,
        allocated_from_incoming: useIncoming,
        remaining_hq_stock_qty_after: remainStock,
        remaining_hq_stock_pending_qty_after: remainIncoming,
      })
    })
  })

  return result
}

const searchTableAndAddClick = async (orders) => {
  try {
    const [tab] = await chrome.tabs.query({ active: true, currentWindow: true })

    if (!tab || !tab.id) {
      alert('활성 탭을 찾을 수 없습니다.')
      return
    }

    console.log('현재 탭 정보:', tab)
    console.log('현재 URL:', tab.url)

    if (!tab.url || !tab.url.includes('supplier.coupang.com')) {
      alert('쿠팡 서플라이어 페이지에서 실행해주세요.\n\n현재 URL: ' + tab.url)
      return
    }

    const response = await chrome.runtime.sendMessage({
      action: 'searchAndClickOrders',
      tabId: tab.id,
      orders: orders,
    })

    console.log('searchAndClick response:', response)

    if (response && response.success) {
      const message =
        `${response.clickedCount}개의 주문을 선택했습니다.\n\n` +
        `전체 페이지: ${response.totalPages || 1}\n` +
        `전체 행 수: ${response.totalRows || 0}`
      alert(message)
    } else if (response && response.error) {
      // 페이지 새로고침 필요한 경우
      if (response.error.includes('새로고침')) {
        const confirmReload = confirm(
          '페이지를 새로고침해야 합니다.\n\n' +
            '(확장 프로그램이 업데이트된 경우 필요합니다)\n\n' +
            '지금 새로고침하시겠습니까?'
        )

        if (confirmReload) {
          await chrome.tabs.reload(tab.id)
          alert('새로고침 완료!\n\n잠시 후 다시 시도해주세요.')
        }
      } else {
        alert('주문 선택 실패: ' + response.error)
      }
    } else {
      alert('주문 선택 실패: 알 수 없는 오류')
    }
  } catch (err) {
    console.error('searchTableAndAddClick 오류:', err)
    alert('주문 선택 중 오류 발생: ' + err.message)
  }
}
